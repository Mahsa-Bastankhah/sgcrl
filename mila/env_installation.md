def reconcile(task_dict):
    # Midpoint and Tolerance derived from current environment and old data
    MIDPOINT = np.array([0.27, 0.0, 0.02])
    TOLERANCE = np.array([0.025, 0.0, 0.0])

    # 1. Transform Absolute Goals to Relative Offsets
    goals = task_dict['goals'] - MIDPOINT
    
    # 2. Convert single-point starts to bounding boxes (min, max)
    # Shape changes from (Tasks, Cubes, 3) -> (Tasks, Cubes, 2, 3)
    starts_min = task_dict['starts'] - TOLERANCE
    starts_max = task_dict['starts'] + TOLERANCE
    starts = np.stack([starts_min, starts_max], axis=2)
    
    return starts, goals
